% ═══════════ 1 ═══════════
\et{1}{La question}
\begin{pts}
\item Les élus du Congrès américain sont \textbf{obligés de déclarer} leurs transactions boursières (loi STOCK, 2012). Deux ETF vendent cette information au public : \textbf{NANC} copie les élus \textbf{démocrates}, \textbf{GOP} les \textbf{républicains}.
\item Ils sont nés le même jour (\textbf{06/02/2023}), coûtent 0,72--0,73 \%/an, et pèsent \textbf{282 M\$} (NANC, 104 lignes) et \textbf{88 M\$} (GOP, 138 lignes).
\item \textbf{Ce qu'on cherche} : (i) peut-on \textbf{reconstruire} leur portefeuille à partir des seules déclarations publiques ? (ii) cette stratégie a-t-elle un \textbf{avantage} sur le marché ?
\item \textbf{Pourquoi ce terrain est bon} : un ETF doit \textbf{publier son portefeuille chaque jour}. On peut donc comparer notre réplique à la vérité, ligne par ligne --- c'est rare.
\end{pts}

% ═══════════ 3 ═══════════
\et{3}{Ce que le fonds dit faire --- et ce qu'il fait vraiment}
\textbf{Ce qu'annonce le prospectus} :
\begin{pts}
\item Univers : les titres achetés par les élus du parti, sur les \textbf{3 dernières années}, via les déclarations.
\item Pondération au \og{}\textbf{niveau de trading}\fg{} : gros achats, achats \emph{répétés}, achats \emph{par plusieurs élus} $\to$ surpondérés ; petits, ventes récentes, coups isolés $\to$ sous-pondérés.
\item Entre \textbf{100 et 200 lignes}. Aucun plafond de position n'est annoncé.
\end{pts}
\textbf{Ce que révèlent les rapports annuels} --- et qu'on avait manqué en ne lisant que le résumé :
\begin{pts}
\item \textbf{Le fonds a changé de méthode.} Son taux de rotation passe de \textbf{62 \% en 2024 à 10 \% en 2025} : il gérait activement, il ne touche presque plus à rien.
\item \textbf{Un événement daté} : le \textbf{02/08/2024} un nouveau gérant (Tidal) reprend et \og{}\emph{eliminated 582 very small positions}\fg{} --- de $\sim$600 lignes à $\sim$170.
\item \textbf{Il y a du jugement humain} : les titres sont \og{}\emph{croisés avec les rôles en commission}\fg{} des élus. Cela n'apparaît dans aucune règle publiée.
\item \textbf{Le prospectus est incomplet} : aucun plafond n'y figure, mais le fonds a \textbf{vendu 15 \% de son NVDA} en un an --- il écrête en pratique.
\end{pts}

\et{4}{Ce qu'on fait --- trois lectures de leur règle}
Leur règle étant qualitative (\og{}niveau de trading\fg{}), on l'a traduite de trois façons. Notation : $A_k$ le montant déclaré, $\delta_k$ la date de divulgation, $P_i$ le cours, $w_i$ un poids, $q_i$ un nombre de parts.
\begin{center}\scriptsize
\begin{tabular}{@{}R{16mm} R{26mm} R{26mm}@{}}
\toprule
 & \textbf{A --- le score} & \textbf{B --- le livre} \\
\midrule
principe & on \textbf{reclasse} tout chaque mois & on \textbf{achète à chaque déclaration}, puis on ne touche plus \\
ventes & ignorées & retirent des parts \\
sélection & les 150 plus gros scores & tout titre déclaré \\
plafond & 8,5 \% & aucun \\
poids & ramenés à la cible & \textbf{dérivent} avec les cours \\
\midrule
sa rotation & 66 \%/an & 13 \%/an \\
\textbf{quel régime} & \textbf{le fonds en 2023-24} (44--62 \%) & \textbf{le fonds en 2025} (10 \%) \\
\bottomrule
\end{tabular}
\end{center}
\[ \text{A}:\quad s_i=\underbrace{B_i}_{\$\ \text{acheté}}\times\underbrace{m_i}_{\text{nb d'élus}},\qquad w_i\propto s_i\ \ (\text{plafonné}) \]
\[ \text{B}:\quad q_i(t)=q_i(t_0)+\!\!\sum_{\text{déclarations}}\!\!\pm\frac{A_k}{P_i(\delta_k)},\qquad w_i=\frac{q_iP_i}{\sum_j q_jP_j} \]
\obs{\textbf{C --- la bascule} : \textbf{A jusqu'au 02/08/2024, puis B}. Ce n'est pas un compromis arbitraire : c'est exactement ce que le fonds a fait, et la rotation le prouve (ligne \og{}quel régime\fg{} ci-dessus). \textbf{C est notre version de référence.}}

\et{6}{Pourquoi ça ne colle pas}
On compare enfin des \textbf{dollars} et non des pourcentages : leur position ($w^{\star}\times$AUM) face au \textbf{\$ total d'achats jamais déclaré} par le parti sur ce titre.
\begin{center}\scriptsize
\begin{tabular}{@{}l r r r@{}}
\toprule
 & leur position & flux déclaré & rapport \\
\midrule

GOOG & 16,4 M\$ & 16,4 M\$ & \textbf{1,0} \\
AMZN & 13,3 M\$ & 13,8 M\$ & \textbf{1,0} \\
MSFT & 14,9 M\$ & 83,0 M\$ & 0,2 \\
\textbf{AMAT} & 14,6 M\$ & 2,2 M\$ & \textbf{6,7} \\

\bottomrule
\end{tabular}
\end{center}
\lire{si le fonds suivait les déclarations, ce rapport serait \textbf{le même partout}. Il va de \textbf{0,2 à 25}. Et les seuls titres où il vaut 1 sont \emph{exactement} ceux qu'on reproduit bien. \textbf{Le fonds détient 25 fois plus de FIX que tout ce que les républicains ont jamais déclaré sur ce titre} : aucune règle fondée sur les déclarations publiques ne peut produire ça.}
\obs{\textbf{Contre-épreuve --- ce n'est pas notre faute.} On a audité et réparé notre propre chaîne : \textbf{7\,287 transactions} écartées par un filtre ont été réintégrées, \textbf{10 séries de prix} manquantes récupérées, les montants \og{}$>$1 M\$\fg{} revalorisés. \textbf{La fidélité ne bouge pas.} On a aussi testé l'hypothèse \og{}ils utilisent les déclarations de \emph{patrimoine} et non de transactions\fg{} : \textbf{réfutée} --- AMAT y pèse 0,009 \% contre 5,19 \% du fonds.}

% ═══════════ 7 ═══════════
\et{7}{Conclusion}
\begin{pts}
\item \textbf{Ces produits ne sont pas reproductibles à partir des données publiques.} On retrouve leurs grands noms et l'ordre de grandeur de leur performance, mais pas leur portefeuille : la moitié pour NANC, moins d'un tiers pour GOP.
\item \textbf{On sait pourquoi, et c'est chiffré} : (i) leurs positions valent jusqu'à \textbf{25 fois} le flux déclaré correspondant --- leur fournisseur de données n'est pas les fourchettes publiques ; (ii) le gérant ajoute un \textbf{jugement discrétionnaire} qu'il admet lui-même ; (iii) le fonds a \textbf{changé de mécanique} en août 2024.
\item \textbf{Et surtout : il n'y a pas d'avantage.} $\beta\approx1$, aucun $t$ au-dessus de 2. Le fonds démocrate bat le S\&P de 2,7 points par an, le républicain le sous-performe de 6,2 : c'est une \textbf{exposition au marché} avec un biais sectoriel, pas une compétence.
\end{pts}
