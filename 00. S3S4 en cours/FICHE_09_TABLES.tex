\section{membres.csv --- 6 familles}
\textbf{A Identité/mandat (13)} : parti, chambre, état, congrès min/max, ancienneté,
\texttt{leader} (+postes, fichier vérifié des 12 postes de parti), comités THOMAS et
\texttt{secteurs\_juridiction} = $\bigcup_c \mathrm{COMITE\_GICS}[c]$ (matrice gelée du 08).

\textbf{B Activité, temps de présence (21)} : $n$ trades (brut/$\mathcal{T}$/achats/ventes),
notional $\sum A_j$ (midpoint --- mesure censurée par intervalle), taille médiane,
$\mathbb{P}(A_j \le 8\,000{,}5)$, délai médian de déclaration, parts de retard ($>45$ j, $>365$ j),
owner dominant, parts conjoint/élu, premier/dernier trade, durée d'activité,
\texttt{actif\_12m} (\og encore dans le jeu\fg{} : 103 membres).

\textbf{C Portefeuille reconstruit (36 ; NaN si non reconstruit)} --- parts FIFO
$n_k^m(t)$, $V_m = \sum_k n_k^m P_k$, $w_k^m = n_k^m P_k / V_m$ :
$$\mathrm{HHI} = \overline{\textstyle\sum_k (w_k^m)^2}, \quad N^{e\!f\!f} = 1/\mathrm{HHI}, \quad
\mathrm{TO} = \frac{\sum_j A_j}{2 \bar V_m}\cdot\frac{252}{T_m}$$
durées de détention par épisodes FIFO : médiane pondérée-parts et médiane de survie
\textbf{Kaplan-Meier} (censure = position encore ouverte ; NaN si $>50\,\%$ encore détenu ---
lire avec \texttt{part\_ouverte}) ; actions principales \texttt{top1..3\_ticker} + parts,
\texttt{top10\_tickers} ; 12 parts sectorielles GICS ($\sum = 1$).

\textbf{D Risque/performance (14 ; série time-weighted $r_m$)} :
$$\texttt{vol} = \sigma(r_m)\sqrt{252}, \qquad \texttt{sharpe} = \tfrac{\bar r_m}{\sigma}\sqrt{252}$$
$$\texttt{sortino} = \tfrac{\bar r_m}{\sigma(r_m^-)}\sqrt{252}, \qquad
r_m = \alpha + \beta r_{SPY} + \varepsilon$$
$$\beta,\ \alpha_{ann},\ \texttt{appraisal} = \tfrac{\alpha}{\sigma_\varepsilon}\sqrt{252},\ t_{appr}$$
plus CAGR, maxDD, IR vs SPY ($+t$), excès de CAGR vs SPY, \texttt{eligible}
($n \ge 10$ trades et $T_m \ge 126$ j : 223 membres).

\textbf{F Track record par trade (7 ; critères du 03)} --- pour chaque achat $j$, entrée
$J{+}1(\tau_j)$, sortie $\min(\text{première vente FIFO}, +252\,\text{j})$, \textbf{brut} vs SPY :
$$\texttt{hit\_rate} = \mathbb{P}(x_j > 0), \qquad t_x = \tfrac{\bar x}{\sigma_x/\sqrt{n}}$$
$$\texttt{profit\_factor} = \min\!\Big(\tfrac{\sum x^+}{|\sum x^-|},\ 10\Big)$$
(58\,926 achats évalués ; excès moyen pooled $-0{,}23\,\%$ --- cohérent avec le
\og copier tout le monde perd\fg{} du 06).

\textbf{E Comportement (3)} : \texttt{part\_ci\_achats} (achats alignés comité$\times$secteur,
$16{,}4\,\%$ sur $\mathcal{T}$), \texttt{part\_trades\_repetes}, \texttt{part\_coachat}
($\ge 3$ acheteurs le même mois).

\section{tickers.csv --- 4 familles}
\textbf{A Identité (8)} : description, classe d'actif, secteur GICS, ETF proxy, délisté (+type).
\textbf{B Flux Congrès (19)} : trades/achats/ventes, notionals (net $=$ buy $-$ sell),
$n$ membres (acheteurs/vendeurs), part Démocrate du notional d'achat, part Sénat, part conjoint,
part alignée comité, bornes de dates, mois actifs, \texttt{coachat\_max}, top-3 membres.
\textbf{C Marché (10)} --- sur la série de prix \textbf{brute} (jours cotés, pas le forward-fill du
moteur, qui biaiserait la vol des délistés vers 0) : $\sigma(\Delta\log P)\sqrt{252}$,
$\beta$ vs SPY ($\ge 126$ obs), CAGR fenêtre, maxDD, ADV\$ $= \operatorname{med}(P^{raw}\cdot\mathrm{Vol})$
(cache volume dédié).
\textbf{D Croisement (3)} : excès moyen à 21/63 j après les achats (entrée $J{+}1(\tau_j)$, brut vs SPY).

\section{membres\_annees.csv --- panel}
Par (membre, année) : activité ($n$ trades, notional, tickers, secteur dominant) et performance
annuelle de $r_m$ (vol, Sharpe, $\texttt{ret\_annee} = \prod(1+r_m)-1$, excès vs SPY mêmes jours ;
NaN si $<40$ jours actifs).

\section{Ce qui garantit les chiffres}
\begin{itemize}
  \item Moteur \textbf{recopié à l'identique} des notebooks audités (05b via 07 ; étiquettes
  \og COPIE EXACTE\fg{} dans le notebook) ; validations : périmètre $118\,029 / 3\,289 / 266 / 223$,
  Khanna ($32\,684$ trades, Sharpe $0{,}87$) et McCaul reproduits, épisodes FIFO
  $61\,963 / 28\,944$ ($44\,\%$ encore détenus), KM pooled $910$ j, Pelosi 2024 $= +38{,}5\,\%$,
  alignement comité $17{,}0\,\%$ (nb) $/ 15{,}2\,\%$ (\$) sur le périmètre du 08.
  \item \textbf{8 contrôles croisés} avant écriture (sommes de trades et de notionals identiques
  entre les deux tables au centime, bipartite $\sum_k n_{membres} = \sum_m n_{tickers}$, bornes de
  dates, compteurs de population) : aucun CSV incohérent ne peut sortir.
  \item Dictionnaire aligné par \texttt{assert} sur les colonnes réellement écrites (145).
\end{itemize}

\section{Limites à garder en tête}

encore en OTC chez Yahoo (ex. BBBY).
