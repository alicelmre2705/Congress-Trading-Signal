\titre{1. Le moteur}{\S{}0--1}
{\small
\begin{gather*}
r_m(t)=\frac{\sum_k n_k^m(t{-}1)\,P_k(t)}{\sum_k n_k^m(t{-}1)\,P_k(t{-}1)}-1 \\[3pt]
\mathrm{lag}_j=\delta_j-\tau_j\quad(\text{délai légal}\le 45\,\text{j})
\end{gather*}}
\begin{itemize}
\item Les portefeuilles ne sont jamais déclarés : seules le sont les transactions.
\item Reconstruction \textbf{trade par trade en FIFO} (premier acheté, premier vendu ; jamais de position négative).
\item Rendement quotidien \emph{time-weighted} : mesure la qualité des choix, pas la taille des apports d'argent.
\end{itemize}

\titre{2. Qui trade, et comment}{\S{}2--4}
{\small
\begin{gather*}
\mathrm{HHI}_m=\operatorname{moy}_t\!\Big(\textstyle\sum_k (w_k^m)^2\Big)\qquad
N_{\mathrm{eff}}=\tfrac{1}{\mathrm{HHI}} \\[3pt]
\mathrm{turnover}_m=\frac{\sum_j A_j}{2\,\bar V_m}\cdot\frac{252}{T_m}
\end{gather*}}
\begin{itemize}
\item Trois mesures par membre : rythme de trading, concentration ($N_{\mathrm{eff}}$ = lignes équivalentes équipondérées), rotation.
\item Exemple \textbf{Pelosi} : \textbf{52\,\% d'achats}, taille médiane \textbf{750\,000\,\$}, délai médian \textbf{26 j}, compte détenu par le conjoint.
\item \textbf{103 membres} encore actifs sur les 12 derniers mois.
\end{itemize}

% ---------------- §3 ----------------
\titre{3. Durées de détention}{\S{}5}
{\small
\begin{gather*}
\hat S_m(t)=\mathbb{P}(\text{part encore détenue après } t\ \text{j}) \\[3pt]
\text{durée médiane}=\inf\{t:\hat S_m(t)\le 0{,}5\}
\end{gather*}}
\begin{itemize}
\item Durée de détention estimée par \textbf{Kaplan-Meier} (courbe de survie).
\item Positions jamais revendues = \emph{censurées}, pas écartées (sinon durées sous-estimées).
\item \textbf{61\,963 épisodes clos} / \textbf{28\,944 ouverts} (44\,\% des parts) ; médiane \textbf{910 j}.
\end{itemize}

% ---------------- §4 ----------------
\titre{4. Risque et performance}{\S{}6--7}
{\small
\begin{gather*}
\text{Sharpe}=\frac{\bar r_m}{\sigma(r_m)}\sqrt{252} \\[3pt]
r_m=\alpha+\beta\,r_{\mathrm{SPY}}+\varepsilon\ \Rightarrow\ \text{appraisal}=\frac{\alpha}{\sigma_\varepsilon}\sqrt{252} \\[3pt]
|t|\ge 2\ \Longrightarrow\ \text{difficilement dû au hasard}
\end{gather*}}
\begin{itemize}
\item Outils standard sur le rendement quotidien : Sharpe, alpha, information ratio.
\item Surtout le \textbf{$t$ de Student} : sans significativité, un rendement élevé ne prouve rien.
\item Volatilité médiane des 223 éligibles : \textbf{20,4\,\%}.
\item Pelosi : excès de croissance \textbf{+3,4\,\%/an}, mais IR 0,31 (\textbf{$t=1{,}06$}) $\to$ non significatif.
\end{itemize}
\fin

% ---------------- CHOIX MÉTHODOLOGIQUES ----------------
\titre{Choix méthodologiques}{justifications}
\begin{itemize}
\item \textbf{Reconstruction FIFO} : les dépôts ne déclarent que des transactions ; portefeuille rejoué achat après achat.
\item \textbf{223 membres} : sous 10 trades / 126 jours actifs, vol et Sharpe annualisés perdent tout sens ; les 372 gardent leur ligne d'identité et de flux.
\item \textbf{Time-weighted} : évalue la qualité des décisions, pas la taille des apports.
\item \textbf{Excès brut de frais} : on mesure l'information du membre, pas le coût d'un réplicateur.
\item \textbf{Deux dates} : $\tau$ reconstruit le portefeuille, $\delta$ mesure la transparence ($\mathrm{lag}=\delta-\tau$).
\item \textbf{Montants par fourchettes} : la loi n'impose que des tranches (1\,001--15\,000\,\$\ldots) ; on retient le milieu.
\end{itemize}
