\begin{center}
{\large\bfseries Construire un portefeuille à partir des déclarations du Congrès}\\[0.5mm]
{\small Quatre méthodes, deux lentilles de mesure --- et ce qui reste quand on a tout contrôlé}
\end{center}
\vspace{0.4mm}\hrule\vspace{0.8mm}
{\footnotesize
\textbf{En bref.} On part de la méthode la plus naïve et on l'améliore pas à pas, en mesurant à chaque étape.
Le gain vient d'un seul geste --- \textbf{pondérer aux dollars déclarés}. Il subsiste, après contrôle du marché,
de la taille, de la valeur et du momentum, un $\alpha$ de \textbf{$+1{,}7$ à $+2{,}3$~\%/an}. Mais le dossier
compte 94 essais : le seuil de bruit est $\mathbf{3{,}01}$ et le meilleur $|t|$ vaut $\mathbf{1{,}99}$.
\textbf{Un $\alpha$ positif et résistant, pas un $\alpha$ démontré} --- et la dernière section dit comment il
pourrait le devenir.}
\vspace{0.8mm}\hrule\vspace{1.2mm}

\begin{multicols}{2}

% ═══════════ PROTOCOLE ═══════════
{\color{bleu}\bfseries 0 $\cdot$ Le protocole, identique pour les quatre}\par\vspace{0.3mm}
\begin{pts}
\item \textbf{Données} : 134\,464 transactions déclarées, 372 élus, 2014--2026. Chaque déclaration donne un titre,
un sens, une date, et une \emph{fourchette} de montant --- on retient le \textbf{milieu} $A_k$.
\item \textbf{On se place à la divulgation} $\delta_k$, jamais à la transaction : c'est la seule date où
l'information est publique, donc la seule où la stratégie est réalisable. \emph{À la date de transaction tous les
chiffres seraient meilleurs --- et faux.}
\item \textbf{Fenêtre} de 3 ans glissants, \textbf{recomposition mensuelle}, exécution au close J+1,
\textbf{buy-and-hold} entre deux coupes. Les deux partis sont traités \textbf{séparément et identiquement}.
\item \textbf{Trois mesures}, de la plus indulgente à la plus sévère :
\end{pts}
\[ \underbrace{x_Y=\!\prod_{t\in Y}\!(1{+}r_t)-\!\prod_{t\in Y}\!(1{+}r^{m}_t)}_{\text{excès brut}}
   \qquad t=\frac{\bar x}{\sigma_x/\sqrt n} \]
\[ \underbrace{r-r_f=\alpha_1+\beta(r^{m}\!-\!r_f)+\varepsilon}_{\text{on retire le marché}} \]
\[ \underbrace{r-r_f=\alpha_4+\beta_{\mathrm M}(r^{m}\!-\!r_f)+\beta_{\mathrm S}\mathrm{SMB}
   +\beta_{\mathrm H}\mathrm{HML}+\beta_{\mathrm U}\mathrm{MOM}+\varepsilon}_{\text{on retire aussi taille, valeur, momentum}} \]
\obs{\textbf{Le seuil qui commande toutes les conclusions.} Ce dossier a produit \textbf{94} bilans distincts. Sous
$H_0$, le maximum de $|t|$ sur $N$ essais vaut en espérance $\sqrt{2\ln N}$, soit $\mathbf{3{,}01}$. \textbf{Un $t$
de 2 ne prouve rien ici} : c'est contre 3,01 qu'il faut le lire, et c'est ce qui évite de se raconter une histoire.}

% ═══════════ M1 ═══════════
\meth{M1}{Une voix par membre}{chaque élu qui achète a le même poids, quel que soit le montant --- l'hypothèse la
plus neutre possible.}
Soit $\mathcal S_t$ les titres achetés par au moins 2 élus sur la fenêtre, et $\mathcal B_m$ ceux achetés par l'élu $m$ :
\[ w_i=\frac1{K_t}\sum_{m}\frac{\mathbf 1\{i\in\mathcal B_m\}}{|\mathcal B_m|},\qquad K_t=\#\{\text{élus acheteurs}\} \]
Chaque élu dispose d'une voix, qu'il répartit entre ses titres.
\begin{center}\scriptsize
\begin{tabular}{@{}l r r r r@{}}
\toprule
 & excès/an & $t$ & $\alpha_1$ & $\alpha_4$ \\
\midrule
toutes chambres & $+0{,}35$~\% & 0,18 & $-0{,}03$ & $+1{,}00$ \\
\bottomrule
\end{tabular}
\end{center}
\lecon{\textbf{l'égalité ne produit rien.} $t=0{,}18$ : le portefeuille est indiscernable du marché
($\beta=1{,}01$). Si l'information vaut quelque chose, ce n'est pas en traitant tous les élus à égalité qu'on
l'extrait. \textbf{Il faut un poids} --- lequel ?}

% ═══════════ M2 ═══════════
\meth{M2}{Pondérer aux dollars déclarés}{celui qui engage 500\,000~\$ y croit plus que celui qui en engage 5\,000.}
\[ B_i=\!\!\sum_{k\,:\,i(k)=i}\!\! A_k,\qquad w_i\propto B_i \]
\begin{center}\scriptsize
\begin{tabular}{@{}l r r r r@{}}
\toprule
 & excès/an & $t$ & $\alpha_1$ & $\alpha_4$ \\
\midrule
NANC & $+3{,}58$~\% & 0,99 & $+0{,}40$ & $+1{,}37$ \\
GOP & $+1{,}51$~\% & 0,58 & $+1{,}19$ & $+2{,}27$ \\
\emph{repère : équipondéré} & \emph{$-0{,}67$~\%} & \emph{$-0{,}58$} & \emph{$-0{,}91$} & \emph{$+0{,}38$} \\
\bottomrule
\end{tabular}
\end{center}
\lecon{\textbf{c'est le geste qui compte.} L'excès passe de $+0{,}35$ à $+3{,}58$~\%/an, et l'équipondéré ---
même univers, poids égaux --- est \textbf{négatif}. Tout l'écart vient donc de la \emph{taille des mises}.
Mais le $\beta$ monte à \textbf{1,13} : une partie est de l'exposition, pas du talent. \textbf{Et le $t$ reste
faible} (0,99) : le portefeuille est trop concentré, donc trop volatil.}

% ═══════════ M3 ═══════════
\meth{M3}{Ajouter le consensus, et borner la concentration}{un titre acheté par dix élus est un signal plus sûr
qu'un titre acheté par un seul --- et aucune ligne ne doit pouvoir emporter le portefeuille.}
Avec $m_i$ le nombre d'élus distincts acheteurs, et $\mathcal U$ les 150 plus gros scores :
\[ s_i=B_i\,m_i,\qquad v_i=\frac{s_i}{\sum_{j\in\mathcal U}s_j},\qquad
   \boxed{\,w_i=\min(c,\ \lambda v_i)\,} \]
$\lambda$ est l'unique réel tel que $\sum_i w_i=1$, avec $c=8{,}5$~\% : un \textbf{remplissage par niveau}, qui
existe et est unique dès que $|\mathcal U|\ge\lceil 1/c\rceil=12$.
\begin{center}\scriptsize
\begin{tabular}{@{}l r r c r@{}}
\toprule
 & excès/an & $t$ & IC 95~\% & $\alpha_4$ \\
\midrule
\textbf{NANC} & $+2{,}98$~\% & \textbf{1,47} & $[-1{,}4;+7{,}4]$ & $+1{,}75$ \\
\textbf{GOP} & $+1{,}24$~\% & \textbf{1,62} & $[-0{,}4;+2{,}9]$ & $+1{,}68$ \\
\midrule
\emph{$m_i$ seuls, sans les \$} & \emph{$+0{,}33$~\%} & \emph{0,33} & --- & \emph{---} \\
\bottomrule
\end{tabular}
\end{center}
\lecon{\textbf{le consensus n'apporte rien tout seul} ($+0{,}33$~\%) --- mais \textbf{il stabilise}. L'excès baisse
($3{,}58\to2{,}98$) et pourtant le $t$ \textbf{monte} ($0{,}99\to1{,}47$), l'IC se resserre de moitié, et le
$\beta$ redescend à 1,08. On échange du rendement contre de la \textbf{fiabilité} --- et c'est le bon échange,
parce que c'est le $t$ qu'on cherche, pas le rendement.}

% ═══════════ M4 ═══════════
\meth{M4}{Écarter le bruit des comptes gérés}{le gérant du fonds dit ignorer les dépôts en masse : \og{}\emph{300
trades \ldots{} 5 shares of something \ldots{} that's just a rebalance}\fg{}. On teste sa règle.}
On écarte toute opération d'un dépôt de $n_k\ge\theta=50$ opérations du même élu le même jour --- soit
\textbf{54~\%} du \$ démocrate et \textbf{58~\%} du républicain.
\begin{center}\scriptsize
\begin{tabular}{@{}l r r c r@{}}
\toprule
 & excès/an & $t$ & IC 95~\% & $\alpha_4$ \\
\midrule
NANC & $+4{,}55$~\% & 1,17 & $[-3{,}9;+13{,}0]$ & $+1{,}75$ \\
GOP & $+1{,}52$~\% & 1,04 & $[-1{,}7;+4{,}7]$ & \textbf{$+2{,}28$} \\
\bottomrule
\end{tabular}
\end{center}
\lecon{\textbf{le meilleur rendement, la pire fiabilité.} L'excès grimpe à $+4{,}55$~\% mais le $t$ \emph{baisse}
($1{,}47\to1{,}17$), l'IC \textbf{double} de largeur et le $\beta$ monte à 1,15. On a acheté du rendement avec de
la variance. \textbf{Le filtre n'est pas retenu} --- et il introduirait un paramètre ($\theta$) de plus.}

\fig{meth_cascade.png}{la cascade, méthode par méthode : excès brut (gris), $\alpha$ marché seul (bleu),
$\alpha$ quatre facteurs (vert). Le passage du gris au bleu mesure ce que le marché explique.}

% ═══════════ LE TEST ═══════════
\meth{$\star$}{Le test qui tranche}{tout l'excès n'est-il qu'une exposition déguisée ?}
\begin{center}\scriptsize
\begin{tabular}{@{}l r r r r@{}}
\toprule
 & $\alpha_4$ & $t$ & $\beta_{\mathrm{SMB}}$ & $\beta_{\mathrm{HML}}$ \\
\midrule
M3 --- NANC & $+1{,}75$ & 1,81 & $-0{,}12$ & $-0{,}13$ \\
M3 --- GOP & $+1{,}68$ & 1,82 & $-0{,}08$ & $+0{,}06$ \\
M4 --- GOP & $+2{,}28$ & \textbf{1,99} & $-0{,}03$ & $+0{,}11$ \\
\emph{le SPY lui-même} & \emph{$+0{,}04$} & \emph{0,12} & \emph{$-0{,}12$} & \emph{$+0{,}01$} \\
\bottomrule
\end{tabular}
\end{center}
\fig{meth_facteurs.png}{les chargements factoriels. La ligne pointillée est le SMB du SPY : nos portefeuilles
sont \emph{à son niveau} --- ils n'ont pas de biais de taille supplémentaire.}
\lecon{\textbf{l'$\alpha$ ne disparaît pas : il augmente} ($+1{,}26\to+1{,}75$). Deux raisons, et aucune n'est
celle qu'on attendait. \textbf{(i)} Il n'y a \emph{pas} de biais de taille : le chargement SMB de nos répliques
($-0{,}12$) est exactement celui du \textbf{SPY} ($-0{,}12$). \textbf{(ii)} Le tilt réel est un tilt
\textbf{croissance} ($\beta_{\mathrm{HML}}<0$ côté NANC) ; la valeur ayant sous-performé, la contrôler
\emph{relève} l'$\alpha$. \textbf{Il reste donc $+1{,}7$ à $+2{,}3$~\%/an que les facteurs n'expliquent pas.}}

% ═══════════ VERDICT ═══════════
\meth{$=$}{La méthode retenue, et pourquoi c'est M3}{on choisit \emph{avant} de regarder les rendements : la plus
parcimonieuse dont le résultat tient sur les \textbf{deux} partis.}
\begin{pts}
\item \textbf{M3}, pas M4 : M4 gagne plus mais son $t$ est plus faible sur les deux camps, son IC double, et il
coûte un paramètre de plus. \textbf{On ne retient pas la méthode qui a le mieux marché, on retient la plus sobre
qui marche des deux côtés.}
\item M3 est la seule dont le $t$ dépasse 1,4 \textbf{et} pour NANC \textbf{et} pour GOP --- $1{,}47$ et $1{,}62$
en excès annuel, $1{,}81$ et $1{,}82$ en $\alpha_4$. La cohérence entre deux échantillons indépendants vaut plus
qu'un chiffre isolé.
\end{pts}
\[ \boxed{\begin{gathered}
s_i=B_i\,m_i\ \text{ sur 3 ans à }\delta,\quad \mathcal U=\text{top-150}\\[0.4mm]
w_i=\min(c,\lambda v_i),\quad c=8{,}5\,\%
\end{gathered}} \]
\fig{meth_nav.png}{les NAV, \textbf{toutes rebasées à 100 au 03/02/2014} --- la seule façon de les comparer à l'œil (échelle log). Sur cette fenêtre commune : M1 \textbf{530}, M2 \textbf{650}, \textbf{M3 672}, M4 \textbf{698}, \textbf{SPY 528}.}

% ═══════════ CE QUI MANQUE ═══════════
{\color{bleu}\bfseries $\neq$ $\cdot$ Ce qui empêche de conclure, et comment le lever}\par\vspace{0.3mm}
\begin{pts}
\item \textbf{Le nombre d'essais.} $|t|_{\max}=1{,}99$ contre un seuil de $3{,}01$. Ce seuil ne vient pas des
données mais de \textbf{nous} : 94 bilans. \emph{Le levier le plus puissant ne coûte aucun calcul} --- déclarer
M3 à l'avance et la tester \textbf{une seule fois} sur une période réservée se lit contre $1{,}96$, pas $3{,}01$.
\item \textbf{Les coûts.} Rien n'est facturé. M3 tourne à $\sim$66~\%/an ; à 10~pb l'aller-retour, cela retire de
l'ordre de $0{,}1$~pt/an --- petit devant $+1{,}75$, mais à chiffrer.
\item \textbf{La période.} 2014--2026 est un marché haussier dominé par la croissance, précisément le tilt de M3.
\item \textbf{Où loge l'$\alpha$ ?} Non testé : par secteur, par taille de dépôt, par élu. Si l'$\alpha_4$ se
concentre sur un sous-ensemble identifiable, c'est un résultat ; s'il est diffus, c'en est un autre --- et les
deux valent d'être sus.
\end{pts}
\obs{\textbf{Ce qu'on peut écrire honnêtement, aujourd'hui} : \emph{pondérer les achats déclarés du Congrès par
les dollars engagés et le nombre d'élus, en bornant la concentration, produit un excès de $+1{,}2$ à
$+3{,}0$~\%/an dont $+1{,}7$~\%/an survit au contrôle du marché, de la taille, de la valeur et du momentum. Sur
94 essais, ce résultat n'atteint pas le seuil de significativité, et une validation hors échantillon est
nécessaire pour trancher.}}

\end{multicols}
