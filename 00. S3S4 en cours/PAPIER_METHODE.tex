% ═══════════ PROTOCOLE ═══════════
{\color{bleu}\bfseries 0 $\cdot$ Le protocole commun --- et ce qui ne l'est pas}\par\vspace{0.3mm}
\begin{pts}
\item \textbf{Données} : 134\,464 déclarations brutes $\to$ \textbf{113\,369 exploitées} (350 élus,
2\,755 titres, 2014--2026). Chaque déclaration donne un titre, un sens, une date et une \emph{fourchette} de
montant --- on retient le \textbf{milieu} $A_k$.
\item \textbf{On se place à la divulgation} $\delta_k$, jamais à la transaction : c'est la seule date où
l'information est publique, donc la seule où la stratégie est réalisable. \emph{À la date de transaction tous les
chiffres seraient meilleurs --- et faux.}
\item \textbf{Le backtest démarre où la spécification est applicable} --- le 28/02/2014. Le plafond de
8,5~\% n'a de solution que si l'univers compte au moins $\lceil 1/c\rceil=12$ titres ; à la première fin de
mois la fenêtre de trois ans ne contient qu'un mois de déclarations et les démocrates n'ont que \textbf{9}
titres. Critère \emph{structurel}, sans aucun rendement --- et il fait \emph{baisser} le résultat.
\item \textbf{Recomposition mensuelle}, exécution au \textbf{close J+1}, \textbf{buy-and-hold} entre deux coupes.
Ce qui n'est pas commun est dans le tableau ci-dessus, plus le \textbf{périmètre} : M1 prend les deux chambres
ensemble sans distinguer les partis, M2--M4 traitent chaque parti \textbf{séparément et identiquement}.
\item \textbf{Long seulement} : la jambe courte, testée à $\tau$ (la date la plus favorable), termine à
\textbf{16} sur base 100 --- c'est le marché à l'envers ($\beta=-1{,}03$).
\item \textbf{Trois mesures}, de la plus indulgente à la plus sévère :
\end{pts}
\[ \underbrace{x_Y=\!\prod_{t\in Y}\!(1{+}r_t)-\!\prod_{t\in Y}\!(1{+}r^{m}_t)}_{\text{excès brut}}
   \qquad t=\frac{\bar x}{\sigma_x/\sqrt n} \]
\[ \underbrace{r-r_f=\alpha_1+\beta(r^{m}\!-\!r_f)+\varepsilon}_{\text{on retire le marché}} \]
\[ \underbrace{r-r_f=\alpha_4+\beta_{\mathrm M}(r^{m}\!-\!r_f)+\beta_{\mathrm S}\mathrm{SMB}
   +\beta_{\mathrm H}\mathrm{HML}+\beta_{\mathrm U}\mathrm{MOM}+\varepsilon}_{\text{on retire aussi taille, valeur, momentum}} \]
\obs{Facteurs Fama-French-Carhart, repris tels quels de la bibliothèque de Kenneth French.}

% ═══════════ M1 ═══════════
\meth{M1}{Une voix par membre}{chaque élu qui achète a le même poids, quel que soit le montant --- l'hypothèse la
plus neutre possible.}
\concret
\begin{pts}
\item \textbf{Chaque fin de mois} $t$, les opérations \textbf{divulguées} dans les 42 derniers jours de bourse :
$\ \mathcal K(t)=\{k:\ t-42<\delta_k\le t\}$. Pour chaque titre, les achats moins les ventes, sans descendre sous
zéro :
\[ B_i=\!\!\sum_{k\in\mathcal K,\ \text{achat}}\!\! A_k,\quad
   S_i=\!\!\sum_{k\in\mathcal K,\ \text{vente}}\!\!\gamma_k A_k,\quad
   \mathrm{Net}_i=\bigl[B_i-S_i\bigr]^{+} \]
\item \textbf{Une vente de titres détenus depuis plus de 2 ans ne compte pas.} On reconstitue les lots en
\emph{premier entré, premier sorti} ; $\gamma_k$ est la part de la vente portant sur des lots \textbf{récents} :
\[ \gamma_k=\frac{\sum_{h\le\bar a}q_h+r_k}{\sum_h q_h+r_k}\in[0,1],\quad \bar a=504\ \text{j}\approx2\ \text{ans} \]
$q_h$ = quantité vendue provenant d'un lot détenu depuis $h$ jours ; $r_k$ = part \textbf{non appariée} (aucun
achat antérieur connu), comptée comme récente --- le choix prudent. \emph{En pratique} : $\gamma=1$ pour 80~\%
des ventes, $\gamma=0$ pour 17~\%.
\item Univers $\ \mathcal S_t=\{i:\ \mathrm{Net}_i>0,\ m_i\ge2\}$. Chaque élu retenu a \textbf{une voix},
partagée entre ses titres $\mathcal B_m$ --- \textbf{le montant n'entre pas}, 5\,000~\$ pèsent autant que
500\,000~\$ :
\[ w_i=\frac1{K_t}\sum_{m}\frac{\mathbf 1\{i\in\mathcal B_m\}}{|\mathcal B_m|},\qquad K_t=\#\{\text{élus acheteurs}\} \]
\item Exécution $\ q_i=w_i\,V(t)/P_i(t{+}1)$, puis plus rien jusqu'à la coupe suivante --- les poids
\textbf{dérivent} avec les cours.
\end{pts}

% ═══════════ M2 ═══════════
\meth{M2}{Pondérer aux dollars déclarés}{celui qui engage 500\,000~\$ y croit plus que celui qui en engage 5\,000.}
\concret
\begin{pts}
\item La fenêtre passe à \textbf{3 ans} et \textbf{les ventes sortent complètement du signal} --- plus de
$\gamma$, plus de solde :
$\ \mathcal K^P(t)=\{k:\ P(k)=P,\ \text{achat},\ t-756<\delta_k\le t\}$, et
$\ B_i=\sum_{k\in\mathcal K^P,\ i(k)=i}A_k$.
\item Le poids est \textbf{proportionnel} à cette somme : $\ w_i\propto B_i$, plafonné à $c=8{,}5$~\%.
\item \textbf{L'univers et le plafond sont figés ici pour M2, M3 et M4.} En les gardant identiques, la
comparaison M2 \emph{vs} M3 isole le seul facteur ajouté, et jamais un effet de sélection.
\end{pts}
\obs{\emph{Et si on nettait les ventes, comme le fait le fonds ?} Testé \textbf{sur M3} : l'excès NANC
\textbf{tombe} de $+2{,}88$ à $+1{,}31$~\%/an. Les écarter n'est pas une paresse, c'est ce qui marche.}

% ═══════════ M3 ═══════════
\meth{M3}{Ajouter le consensus, et borner la concentration}{dix acheteurs valent mieux qu'un --- et aucune ligne
ne doit pouvoir emporter le portefeuille.}
\concret
\begin{pts}
\item On \textbf{multiplie} les dollars par le \textbf{nombre d'élus distincts} $m_i$. \emph{Le geste se lit à
somme égale} : 200\,000~\$ déclarés par \textbf{deux} élus donnent un score \textbf{double} de 200\,000~\$
déclarés par un \textbf{seul}. On préfère l'argent dispersé à l'argent concentré.
\[ s_i=B_i\,m_i,\qquad m_i=\#\{m(k):k\in\mathcal K^P,\ i(k)=i\} \]
\item \emph{Conséquence, qu'il faut voir} : si $K$ élus déclarent chacun le même montant $a$, alors $B_i=Ka$ et
$s_i=K^2a$ --- le score croît comme le \textbf{carré} du nombre d'élus, car $m_i$ est déjà dans $B_i$ (plus
d'acheteurs, plus de dollars) et on le remultiplie.
\item Univers $\mathcal U(t)$ = les \textbf{150 plus gros scores} ;
$\ v_i=s_i\big/\sum_{j\in\mathcal U}s_j$. On \textbf{plafonne chaque ligne à $c=8{,}5$~\%}, l'excédent
redistribué au prorata jusqu'à ce que plus rien ne dépasse. Le point d'arrivée s'écrit d'un coup :
\[ \boxed{\,w_i=\min(c,\ \lambda v_i)\,}\qquad \lambda\ \text{tel que}\ \sum_i w_i=1 \]
$\lambda\mapsto\sum_i\min(c,\lambda v_i)$ croît de 1 à $|\mathcal U|c$ : $\lambda$ existe et est \textbf{unique}
dès que $|\mathcal U|\ge\lceil 1/c\rceil=12$ titres.
\end{pts}
\obs{\emph{La frontière du top-150 n'est pas nette} --- les montants étant des fourchettes, \textbf{104 titres}
partagent le score du 150\textsuperscript{e} à une coupe de 2014, et \textbf{36} s'échangent selon l'ordre de tri. Mesuré : les trier de A à Z ou de Z à A déplace
l'excès de \textbf{0,03 pt/an}.}

% ═══════════ M4 ═══════════
\meth{M4}{Écarter le bruit des comptes gérés}{le gérant du fonds dit ignorer les dépôts en masse : \og{}\emph{300
trades \ldots{} 5 shares of something \ldots{} that's just a rebalance}\fg{}. On teste sa règle.}
\concret
\begin{pts}
\item \textbf{Avant tout calcul}, on compte les opérations déclarées par un même élu \textbf{le même jour},
$\ n_k=\#\{k':\ m(k')=m(k),\ \delta_{k'}=\delta_k\}$, et l'on écarte celles des dépôts de \textbf{50 opérations
ou plus} --- signature d'un compte piloté par un courtier, pas d'une décision : $\ n_k<\theta=50$.
\item Cela retire \textbf{54~\%} du \$ démocrate et \textbf{58~\%} du républicain. Tout le reste est identique
à M3.
\end{pts}

% ═══════════ ÉPREUVE 3 ═══════════
\meth{\textdollar}{Épreuve 2 --- que coûte l'exécution, et faut-il ralentir ?}{tous les chiffres précédents sont
bruts. On facture le passage d'ordre, puis on demande s'il vaut mieux freiner pour en payer moins.}
\concret
\begin{pts}
\item \textbf{La vitesse} se mesure à chaque coupe :
$\ \mathrm{TO}_t=\tfrac12\sum_i\bigl|w^{\text{réalisé}}_i-w^{\text{dérivé}}_i\bigr|$, la fraction du portefeuille
qui change de mains. \textbf{Elle ne se règle pas, elle découle de la fenêtre} : à chaque coupe celle-ci avance
d'un mois, or 42 jours n'en contiennent que \textbf{deux} --- la moitié du signal est renouvelée et le classement
se refait --- quand 3 ans en contiennent \textbf{trente-six}, dont un seul change. D'où l'écart mesuré :
\textbf{59~\%} de rotation par mois pour M1, \textbf{5~\%} pour M2--M4.
\item \textbf{Les frais} : $\ V_t\leftarrow V_t(1-2\,\mathrm{TO}_t\,c)$ avec $c=10$~pb. Le \textbf{facteur 2}
parce que $\mathrm{TO}_t$ est le montant acheté, égal au montant vendu : on paie des deux côtés.
\end{pts}

(w^{\text{cible}}-w^{\text{dérivé}})$. Le frein n'agit que si la rotation voulue \emph{dépasse} $\tau_{\max}$ --- sinon $\lambda_R=1$ et rien ne
change. À $\tau_{\max}=25\,\%$ il se déclenche donc \textbf{à chaque coupe} sur M1 (59~\%/mois, on ne parcourt

descendre bien plus bas.}

il est \emph{encadré} de plafonds qui échouent ($-0{,}06$ à $10\,\%$, $-0{,}42$ à $2\,\%$). Serrer une
contrainte devrait agir dans un sens, pas alterner : ce profil est celui du bruit.}
