{\color{bleu}\bfseries\footnotesize LA SPÉCIFICATION}\vspace{0.5mm}\hrule\vspace{1mm}

\s{1}{Le trade}
\[ x_k = (\, m_k,\; i_k,\; \tau_k,\; d_k,\; A_k \,) \qquad d_k \in \{-1,+1\} \]
\g{$m_k$ élu $\cdot$ $i_k$ titre $\cdot$ $\tau_k$ transaction $\cdot$ $A_k$ milieu de fourchette.
$\mathcal{C}$ = calendrier de bourse, $n(t)$ = indice de $t$.}

\s{2}{L'univers}
\[ \mathcal{U}_t = \bigl\{\, i : \text{action} \;\wedge\; \exists\, u \in [\,n(t)-4,\, n(t)\,] \text{ tq } P_i(u) \text{ obs.} \,\bigr\} \]
\g{La 2\ieme{} condition dit \textbf{deux} choses : le titre cote \emph{déjà} et cote \emph{encore}.
Le cache prolonge au dernier cours connu $\Rightarrow$ sans elle, une société morte en 2019 cote encore en 2026.}

\s{3}{Dater les lots (FIFO)}
\[ s_k = \min\{ u \in \mathcal{C} : u \ge \tau_k \} \qquad q_k = \frac{A_k}{P_{i_k}(s_k)} \]
\[ h_\ell = \#\bigl(\mathcal{C} \cap (\,s_\ell,\, s_k\,]\bigr) \]
\g{Un achat empile $(s_k,q_k)$ ; une vente consomme les lots les plus anciens.
$h_\ell$ = durée de détention de la tranche $\ell$ vendue par $k$.
\textbf{Le portefeuille des membres ne sert qu'à ça.}}

\s{4}{La purge}
\[ \bar a = 504 \text{ j} \qquad
\gamma_k = \frac{\sum_{\ell \subset k} \mathbf{1}\{h_\ell \le \bar a\}\, q_\ell}{\sum_{\ell \subset k} q_\ell} \in [0,1] \]
\g{Part de la vente $k$ portant sur des lots de moins de 2 ans.
Achat $\Rightarrow \gamma_k = 1$ $\cdot$ vente non appariée $\Rightarrow \gamma_k = 1$ (indatable, on ne l'invente pas vieille).}

\s{5}{Le flux par ticker}
\[ W = 42 \text{ j} \qquad \mathcal{K}_{i,t} = \{\, k : i_k = i,\; t - W < s_k \le t \,\} \]
\[ Buy_{i,t} = \!\!\sum_{k \in \mathcal{K}_{i,t}}\!\! A_k \mathbf{1}\{d_k = {+}1\}
\qquad
Sell^{\varphi}_{i,t} = \!\!\sum_{k \in \mathcal{K}_{i,t}}\!\! A_k \gamma_k \mathbf{1}\{d_k = {-}1\} \]
\[ NetBuy^{\varphi}_{i,t} = \bigl[\, Buy_{i,t} - Sell^{\varphi}_{i,t} \,\bigr]^{+} \]
\[ nBuyers_{i,t} = \#\{\, m_k : k \in \mathcal{K}_{i,t},\; d_k = {+}1 \,\} \]

\s{6}{Le signal}
\[ S_t = \bigl\{\, i \in \mathcal{U}_t : NetBuy^{\varphi}_{i,t} > 0 \;\wedge\; nBuyers_{i,t} \ge 2 \,\bigr\} \]

\s{7}{Les poids --- une voix par membre}
\[ B_{m,t} = \{\, i \in S_t : m \text{ a acheté } i \,\} \]
\[ \mathcal{M}_t = \{\, m : B_{m,t} \ne \emptyset \,\}, \qquad K_t = |\mathcal{M}_t| \]
\[ \boxed{\; w_{i,t} = \frac{1}{K_t} \sum_{m \in \mathcal{M}_t} \frac{\mathbf{1}\{ i \in B_{m,t} \}}{|B_{m,t}|} \;} \]
\g{Chaque membre reçoit $1/K_t$ et le répartit également entre ses titres.
\textbf{$A_k$ n'apparaît pas} : 11 M\$ et 15 k\$ pèsent pareil.}

\cadre{vert}{Trois propriétés, zéro paramètre.}{
$\sum_i w_{i,t} = \frac{1}{K_t}\sum_m \sum_{i \in B_m} \frac{1}{|B_m|} = 1$ --- \emph{normalisation automatique} \\[0.6mm]
$|B_m| = 1\,\forall m \Rightarrow w_i \propto b_i$ \; ; \; $B_m = S_t\,\forall m \Rightarrow w_i = 1/|S_t|$
--- \emph{la règle se place seule entre les deux} \\[0.6mm]
trust rééquilibrant 133 lignes $\Rightarrow 1/(133 K_t)$ par ligne --- \emph{la délégation se dilue toute seule}}

\s{8}{Le portefeuille}
\[ nbr_i = \frac{V(t_R{+}1)\, w_{i,t_R}}{P_i(t_R{+}1)} \qquad V(t) = \!\!\sum_{i \in S_{t_R}}\!\! nbr_i\, P_i(t) \]
\[ r(t) = \frac{V(t)}{V(t{-}1)} - 1 = \sum_i w_i(t{-}1)\, r_i(t) \]
\g{Coupe $t_R$ = dernier jour du mois, exécution au close du jour suivant ; buy-and-hold entre deux coupes.}

\s{9}{L'évaluation}
\[ x_Y = \prod_{t \in Y}(1{+}r) - \prod_{t \in Y}(1{+}r_{\mathrm{SPY}}) \qquad t = \frac{\bar x}{\sigma(x)}\sqrt{n} \]
\[ r - r_f = \alpha + \beta\,(r_{\mathrm{SPY}} - r_f) + \varepsilon \qquad \alpha_{\mathrm{ann}} = 252\,\alpha \]
