\section{Reconstruction du portefeuille d'un membre}
\begin{itemize}
  \item \textbf{Achat} $\to$ parts, au 1\ier{} jour coté $\ge$ la transaction :
        $q = \dfrac{\text{montant}}{P_i(\tau)},\ \tau=\min\{t\ge \code{traded}\}$.
  \item \textbf{Vente} $\to$ retire des parts (short interdit) :
        $q^- = \min(\text{montant}/P_i,\ \text{parts})$.
  \item \textbf{Détentions cumulées} :
        $n_i(t) = \sum_{\text{achats}\le t} q - \sum_{\text{ventes}\le t} q^-$.
  \item \textbf{Valeur} : $V(t) = \sum_i n_i(t)\,P_i(t)$.
\end{itemize}

\section{Rendement d'un membre (\emph{time-weighted})}
On valorise les parts de la \textbf{veille} $\Rightarrow$ un apport n'est jamais un rendement :
$$r_P(t) = \frac{\sum_i n_i(t-1)P_i(t)}{\sum_i n_i(t-1)P_i(t-1)} - 1$$
\begin{itemize}
  \item Jour sans position : $r_P(t)=0$ (système ouvert) ; winsorisation $\pm 50\,\%$/jour.
  \item $\mathrm{NAV}(t)=\prod_{s\le t}(1+r_P(s))$ = la \textbf{performance pure} (contrairement à $V(t)$
        qui mélange les apports) ; $\mathrm{vol}=\sigma(r_P)\sqrt{252}$,
        $\mathrm{Sharpe}=\tfrac{\overline{r_P}}{\sigma(r_P)}\sqrt{252}$.
\end{itemize}

\textbf{Aperçu de toute la population.} On calcule la série de rendement de \emph{chaque} membre une
seule fois (réutilisée partout) ; cumulative $\Rightarrow$ la tronquer à une date ne regarde jamais le futur.

\section{Qui est éligible ?}
Un ratio de performance n'a de sens que sur assez de matière (un IR sur 3 trades explose). On ne garde que :
$$n_{\text{trades}} \ge 10 \quad\text{et}\quad n_{\text{jours}} \ge 126\ (\sim 6\ \text{mois}).$$
Il reste \textbf{223 éligibles} sur 266.

\fig{fig_rap_elig.png}{0.95}{Qui passe la porte : $n_{\text{trades}}$ (log) $\times$ $n_{\text{jours}}$ ;
la boîte d'éligibilité (coin haut-droit) garde 223 membres sur 266.}

\section{Qui bat le SPY --- talent ou style ?}
Deux critères, testés de la même façon (test $t$ unilatéral, seuil $1{,}645$).

\textbf{Information Ratio} (suppose $\beta=1$) :
$$x(t)=r_P(t)-r_{SPY}(t),\ \ \mathrm{IR}=\frac{\overline{x}}{\sigma(x)}\sqrt{252},\ \ t=\mathrm{IR}\sqrt{n/252}$$

\textbf{Appraisal} (talent, marché retiré) --- régression $r_P=\alpha+\beta\,r_{SPY}+\varepsilon$ :
$$\beta=\frac{\operatorname{Cov}(r_P,r_{SPY})}{\operatorname{Var}(r_{SPY})},\ \
  \alpha=\overline{r_P}-\beta\,\overline{r_{SPY}}$$
$$\text{appraisal}=\frac{\alpha}{\sigma(\varepsilon)}\sqrt{252},\ \
  t_{\text{appr}}=\text{appraisal}\sqrt{n/252}$$
$\beta=1 \Rightarrow$ appraisal $=$ IR ; sinon l'IR mélange \textbf{exposition} (style) et \textbf{talent}.
\emph{Sur $M=223$ tests corrélés, $\sim$11 dépassements du seuil sont attendus par pur hasard --- borne basse.}

\fig{fig_rap_regdemo.png}{0.82}{Régression du membre-exemple : pente $=\beta$, ordonnée $=\alpha$ ;
la droite grise est l'hypothèse $\beta=1$ de l'IR.}

\section{Choisir qui copier chaque année (walk-forward)}
À chaque fin d'année $Y$, on classe les éligibles par un critère, on garde les significatifs, on suit
le top-$K$ en $Y{+}1$ --- point-in-time, aucun regard futur. Classement $=$ les éligibles à
$t_{\text{crit}}>1{,}645$, triés par le critère :
$$\mathrm{classement}(Y)=\{\text{éligibles} \wedge t_{\text{crit}}>1{,}645\}.$$

\section{La combinaison : buy-and-hold}
« Copier et \textbf{tenir} » laisse les poids \textbf{dériver} avec la performance ; les figer reviendrait
à rebalancer chaque jour (faux) :
$$\mathrm{NAV}_k(t)=\!\prod_{s\le t}\!(1+r_{m_k}(s)),\quad N(t)=\sum_k w_k\,\mathrm{NAV}_k(t),$$
$$r_{\text{strat}}(t)=\frac{N(t)}{N(t-1)}-1.$$

\section{Combien copier ? (le choix de $K$)}
On construit les stratégies pour tout $K\in[4,20]$, puis on choisit $K$ \textbf{honnêtement} (sans regard
futur), avec \textbf{le critère de la sélection lui-même} :
$$K^*(Y)=\arg\max_K \operatorname{score}_{\text{crit}}\big(\text{strat-}K\big|_{\le Y}\big).$$
$K=4$ par défaut sans historique.

\section{Verdict}
\begin{itemize}
  \item Le classement est crédible et outillé, mais une fois le marché retiré, le surnombre de « talents »
        \textbf{fond au niveau du hasard} (20 $\to$ 12 $\approx$ 11 attendus) : c'était du $\beta$.
  \item La copie des meilleurs \textbf{ne bat pas le SPY} : top-4 $+3{,}1\,\%$/an mais $t=0{,}94$ et
        $\beta=1{,}20$ $\Rightarrow$ $\alpha<0$. Aucun $K$ (4 à 20), fixe ou walk-forward, ne franchit le seuil.
  \item Sélectionner sur le talent $\beta$-propre neutralise le style ($\beta\approx1$) \textbf{mais rend
        l'excès négatif} : pas de persistance du talent.
  \item Tout ceci est un \textbf{plafond théorique} (entrée à la date de transaction, sans frais) ; la
        version jouable (entrée à la publication + coûts) est traitée dans le notebook~06.
\end{itemize}
