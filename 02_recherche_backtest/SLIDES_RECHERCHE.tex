\begin{frame}{1.1 · La donnée, et ce qui est observable à la date $t$}
  \begin{columns}[T]
    \begin{column}{0.47\textwidth}
      \obs{une déclaration est un enregistrement à \textbf{huit champs}}
      \vspace{-1mm}
      {\footnotesize\[ x_k = \bigl(\,m_k,\ i_k,\ p_k,\ c_k,\ \tau_k,\ \delta_k,\ d_k,\
                       A_k\,\bigr) \]}
      \vspace{-1mm}
      \obs{$m_k$ l'élu · $i_k$ le ticker · $p_k$ le parti · $c_k$ la chambre}
      \obs{$\tau_k$ la date de \textbf{transaction} · $\delta_k$ la date de
       \textbf{divulgation} --- \textbf{27 jours} d'écart médian}
      \obs{$d_k \in \{+1,\,-1\}$ : achat ou vente}
      \obs{$A_k$ : le dépôt ne donne qu'une \textbf{fourchette} de montant --- on en retient le
       \textbf{milieu}}
    \end{column}
    \begin{column}{0.50\textwidth}
      \obs{$\mathcal F_t$ : les déclarations déjà \textbf{publiées} à $t$. Sur une fenêtre de $W$
       jours, ce que le titre $i$ retient}
      \vspace{-1mm}
      {\footnotesize\[ \mathcal K_{i,t} = \bigl\{\, k \in \mathcal F_t :\ i_k = i,\
                       t - W \leq t^{\bullet}_k \leq t \,\bigr\} \]}
      \vspace{-1mm}
      \obs{$t^{\bullet}_k$ est $\tau_k$ \emph{ou} $\delta_k$, jamais les deux. À $\tau$
       l'information n'est pas encore publique : c'est un \textbf{plafond théorique}, pas une
       stratégie qu'on peut tenir}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{1.2 · Séparer une vente passive d'une vente informée\\
  {\normalsize\mdseries une vente de titres achetés il y a plus de deux ans n'entre pas dans le signal}}
  \begin{columns}[T]
    \begin{column}{0.49\textwidth}
      \et{1}{Les achats entrent --- la pile}
      {\scriptsize un dépôt donne un \textbf{montant}, pas des parts : on convertit au prix du
       jour, $q = A_k / P_{i_k}(\tau_k)$. Chaque achat, par \textbf{(élu, titre)}, dépose alors
       une ligne $(\tau_\ell,\ q_\ell)$ --- \emph{une date, un nombre de parts}. La pile ne fait
       que grandir.}\\[3mm]
      \et{2}{Les ventes sortent --- les plus vieilles lignes d'abord}
      {\scriptsize une vente $k$, à la date $\tau_k$, retire $Q$ parts. Elle les prélève sur les
       lignes \textbf{par date d'achat croissante} : $x_\ell$ parts sur la ligne $\ell$, jusqu'à
       $\textstyle\sum_\ell x_\ell = Q$.}
    \end{column}
    \begin{column}{0.49\textwidth}
      \et{3}{Chaque sortie connaît son âge}
      {\scriptsize la ligne entamée porte sa date d'achat, la vente porte la sienne :
       $a_\ell = \tau_k - \tau_\ell$, en jours de bourse. Une même vente peut sortir du vieux
       \emph{et} du récent.}\\[2.5mm]
      \et{4}{$\gamma_k$, la part récente de la vente}
      {\footnotesize\[ \boxed{\ \gamma_k = \frac{1}{Q}\sum_{\ell} x_\ell\
                        \mathbf 1\{a_\ell \leq \bar a\} \ \in [0,1]\ } \]}
      \vspace{-1.5mm}
      {\scriptsize avec $\bar a = 504$ j $\simeq$ 2 ans de bourse.}\\[1.5mm]
      \obs{\textbf{un scalaire par vente} : $1$ = que du récent, $0$ = que du vieux, la
       proportion exacte entre les deux}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{1.4 · Le signal ticker et la pondération}
  \begin{columns}[T]
    \begin{column}{0.49\textwidth}
      \et{1}{Ce qui entre}
      {\scriptsize sur une fenêtre de $W$ jours, pour chaque titre : tout ce qui a été acheté,
       moins ce qui a été vendu \textbf{au prorata de son $\gamma$}.}
      \vspace{-1mm}
      {\footnotesize\[ B_{i,t} = \sum_{k \in \mathcal K_{i,t},\ d_k = +1} A_k, \qquad
                       B^{\gamma}_{i,t} = \sum_{k \in \mathcal K_{i,t},\ d_k = -1} \gamma_k A_k \]}
      \vspace{-1mm}
      {\scriptsize ce qui reste est le \textbf{net acheté},
       $N_{i,t} = B_{i,t} - B^{\gamma}_{i,t}$.}\\[2.5mm]
      \et{2}{Ce qu'on garde}
      {\scriptsize deux conditions à la fois : le net est positif, et \textbf{au moins deux élus
       distincts} l'ont acheté --- un seul acheteur ne fait pas un signal.}
      \vspace{-1mm}
      {\footnotesize\[ S_t = \{\, i :\ N_{i,t} > 0 \ \wedge\ m_{i,t} \geq 2 \,\} \]}
    \end{column}
    \begin{column}{0.49\textwidth}
      \et{3}{Une voix par élu}
      {\scriptsize chacun des $K_t$ élus ayant acheté un titre de $S_t$ reçoit $1/K_t$ du capital,
       et le répartit \textbf{également} entre ses titres. Le poids d'un titre est la somme de ce
       que chaque élu lui a donné.}
      \vspace{-1mm}
      {\footnotesize\[ w_{i,t} = \frac{1}{K_t}\sum_m
                   \frac{\mathbf 1\{i \in \mathcal B_{m,t}\}}{|\mathcal B_{m,t}|} \]}
      \vspace{-1mm}
      {\scriptsize $\mathcal B_{m,t}$ : les titres de $S_t$ achetés par l'élu $m$ ·
       $m_{i,t} = \#\{m_k : k \in \mathcal K_{i,t},\ d_k = +1\}$, les élus acheteurs de $i$ ·
       $\sum_i w_{i,t} = 1$.}\\[2.5mm]
      \et{4}{Ce que ça implique}
      \obs{$A_k$ n'apparaît \textbf{pas} dans $w$ : l'argent décide \emph{si} un titre entre ---
       le signe du net --- \textbf{jamais combien il pèse}}
      \obs{27 élus actifs, donc 3,70~\% chacun : qui a acheté un titre y met tout, qui en a
       acheté onze met 0,34~\% sur chacun}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{1.5 · Choisir la fenêtre $W$ (test de concentration)}
  \obs{on teste 4 fenêtres et on tranche sur la \textbf{concentration}, jamais sur le rendement}
  \vspace{-1mm}
  {\footnotesize\[ W^\star = \min\bigl\{\, W :\ \mathbb P\bigl[\max_i w_{i,t} > 10\,\%\bigr]
     \leq 20\,\% \,\bigr\} \]}
  \vspace{-1mm}
  \lire{$\max_i w_{i,t}$ : poids de la plus grosse ligne, un mois donné (règle : $> 10~\%$ dans
   $\leq 20~\%$ des mois).}
  \par\vspace{1.5mm}
  \centering
  {\footnotesize\setlength{\tabcolsep}{8pt}
  \begin{tabular}{@{}l r r r r@{}}
    \toprule
    $W$ & 21 j & \textbf{42 j} & 63 j & 126 j\\
    \midrule
    $|S_t|$ médian      & 28      & \textbf{64}      & 91     & 156\\
    $K_t$ médian        & 19      & \textbf{32}      & 39     & 53\\
    $\max_i w_{i,t}$ médian & 11,0~\% & \textbf{7,2~\%} & 6,3~\% & 5,4~\%\\
    mois $> 10~\%$      & 58~\%   & \textbf{17~\%}   & 9~\%   & 5~\%\\
    \bottomrule
  \end{tabular}}
  \par\vspace{1.5mm}
  \raggedright
  \lire{$|S_t|$ = titres retenus/mois ; $K_t$ = membres actifs/mois ; dernière ligne = la
   fréquence de sur-concentration, \emph{le critère}.}
  \par\vspace{1.5mm}
  \sortie{$W = 42$ j retenu --- critère et constantes fixés \textbf{avant} de regarder la
   performance ; et cette fenêtre \textbf{ne survit pas} : M2 à M4 tournent sur les 3 ans du
   prospectus (page 19).}
  \vspace{0.5mm}
  \obs{\textbf{ce paramètre sera stress-testé avant toute mise en production} --- rejouer les
   quatre fenêtres \emph{sur le rendement} et publier le balayage entier, comme celui de $\theta$
   page 21 ; faire varier les deux constantes du critère et vérifier que $W^\star$ ne bouge pas ;
   refaire le choix sur la première moitié de l'échantillon pour l'appliquer à la seconde}
\end{frame}

\begin{frame}{1.7 · \ldots et si on ajoute un \textbf{cap de rotation} ? --- à $\tau$}
  \begin{columns}[T]
    \begin{column}{0.49\textwidth}
      \et{1}{Les poids dérivent}
      {\scriptsize entre deux coupes les \textbf{parts sont figées}, donc les poids s'écartent de
       la cible. $w_{i,t}$ la \textbf{cible} · $w^{\text{dérivé}}_{i,t}$ à l'arrivée ·
       $w^{\text{réalisé}}_{i,t}$ au départ.}\\[1.5mm]
      \et{2}{La rotation voulue}
      \vspace{-1mm}
      {\footnotesize\[ \mathrm{TO}^{\text{cible}}_t = \tfrac12 \sum_i
                       \bigl| w_{i,t} - w^{\text{dérivé}}_{i,t} \bigr| \]}
      \vspace{-2mm}
      \et{3}{Le frein}
      \vspace{-1mm}
      {\footnotesize\begin{align*}
         w^{\text{réalisé}}_{i,t} &= w^{\text{dérivé}}_{i,t}\\[-0.5mm]
           &+ \min\bigl(1,\ \overline{\mathrm{TO}}/\mathrm{TO}^{\text{cible}}_t\bigr)\,
              \bigl(w_{i,t} - w^{\text{dérivé}}_{i,t}\bigr)
       \end{align*}}
      \vspace{-3mm}
      {\scriptsize avec $\overline{\mathrm{TO}} = 25~\%$/mois.}
    \end{column}
    \begin{column}{0.48\textwidth}
      \obs{la rotation \textbf{réalisée}, celle qu'on facture, vaut alors
       $\mathrm{TO}_t = \tfrac12\sum_i|w^{\text{réalisé}}_{i,t} - w^{\text{dérivé}}_{i,t}|
       = \min(\mathrm{TO}^{\text{cible}}_t,\ \overline{\mathrm{TO}})$ --- \textbf{le frein tronque
       la rotation à 25~\%}}
      \vspace{2.5mm}
      {\scriptsize\setlength{\tabcolsep}{3.4pt}
      \begin{tabular}{@{}l r r r r r@{}}
        \toprule
        run & excès \%/an & $t$ & $\alpha$ \%/an & $\beta$ & NAV\\
        \midrule
        la stratégie & $+1{,}71$ & 1,03 & $+1{,}05$ & 1,02 & 622\\
        \textbf{$+$ cap 25 \%} & $\mathbf{+0{,}27}$ & \textbf{0,23} & $-0{,}16$ & 1,01 & \textbf{531}\\
        \bottomrule
      \end{tabular}}\par\vspace{3mm}
      \obs{le cap ramène la rotation de 59 à 25~\%/mois --- et coûte \textbf{1,4 point d'excès par
       an}, \emph{sans même compter les frais}}
    \end{column}
  \end{columns}
\end{frame}
