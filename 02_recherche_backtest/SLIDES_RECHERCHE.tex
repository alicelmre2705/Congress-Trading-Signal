\begin{frame}{Le geste : le consensus compte \textbf{au carré}\\
  {\normalsize\mdseries $s_i = B_i \times m_i$ \;--- les dollars, multipliés par le nombre d'élus distincts}}
  \centering
  \vspace{2mm}
  \resizebox{0.95\textwidth}{!}{%
  \begin{tikzpicture}[node distance=9mm]
    \node[noeud=grisC, text width=4.6cm] (a)
      {\textbf{un} élu déclare 200\,000~\$\\[0.6mm]$B_i = 200\,000$, $m_i = 1$\\[0.6mm]
       score $= \mathbf{200\,000}$};
    \node[noeud=helec, text width=4.6cm, right=of a] (b)
      {\textbf{deux} élus déclarent 100\,000~\$\\[0.6mm]$B_i = 200\,000$, $m_i = 2$\\[0.6mm]
       score $= \mathbf{400\,000}$};
    \node[noeud=okgreen, text width=4.6cm, right=of b] (c)
      {\textbf{$K$} élus au même montant $a$\\[0.6mm]$B_i = Ka$, $m_i = K$\\[0.6mm]
       score $= \mathbf{K^2 a}$};
    \draw[arr] (a)--(b); \draw[arr] (b)--(c);
  \end{tikzpicture}}\\[4mm]
  {\footnotesize Le nombre d'élus est \emph{déjà} dans les dollars --- plus d'acheteurs, plus de
   dollars --- et on le remultiplie : le score croît comme le \textbf{carré} du consensus.}\\[3mm]
  \defl{tout le reste est identique à la stratégie 2 : même fenêtre, mêmes achats seuls, même
   univers de 150 titres, même plafond, même exécution. \textbf{Une seule ligne de code change.}}
\end{frame}
