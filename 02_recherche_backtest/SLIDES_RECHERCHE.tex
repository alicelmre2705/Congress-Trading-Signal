\begin{frame}{A2 · Le frein, et le choix de la fenêtre $W$}

        \toprule

      \defl{un seul plafond passe, \emph{encadré} de plafonds qui échouent : c'est le profil du
       bruit. Non retenu.}
    \end{column}
    \begin{column}{0.49\textwidth}
      {\footnotesize\textbf{La fenêtre $W$} --- choisie sur la concentration :}\\[2mm]
      {\scriptsize\setlength{\tabcolsep}{3.4pt}
