\chapdivider{Étape B}{À quoi ressemble un portefeuille}{on ne déclare que des transactions ; on reconstruit le reste}

\begin{frame}{Un portefeuille se reconstruit trade par trade (FIFO)}
  \begin{columns}[T]
    \begin{column}{0.44\textwidth}
      \small
      \textbf{Achat} / \textbf{vente} du titre $i$, montant $a$ :\\[-4mm]
      \begin{align*}
        q &= \frac{a}{P_i(\tau)}\qquad\text{(achat)}\\
        q^- &= \min\!\Bigl(\tfrac{a}{P_i(\tau)},\, n_i(\tau^-)\Bigr)\quad\text{(vente)}
      \end{align*}
      \textbf{Positions et valeur} :\\[-4mm]
      \begin{align*}
        n_i(t) &= \textstyle\sum_{\text{achats}} q-\sum_{\text{ventes}} q^-\\
        V(t)   &= \textstyle\sum_i n_i(t)\,P_i(t)
      \end{align*}
      \defl{$\tau$ = 1\up{er} jour coté $\ge$ date du trade ; $P_i(t)$ = prix ajusté (dividendes, splits). $V(t)$ saute à chaque apport → figure.}\\[1mm]
      \choixbox{FIFO (premier acheté, premier vendu), position courte interdite.}
\end{column}
    \begin{column}{0.54\textwidth}\centering
      \includegraphics[width=\linewidth,height=0.72\textheight,keepaspectratio]{figs_bt/B1_vt_membre.png}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Le rendement « en parts » : un apport n'est jamais une performance}
  \begin{columns}[T]
    \begin{column}{0.44\textwidth}
      \small
      \textbf{Rendement time-weighted} (parts de la veille), puis \textbf{NAV} :\\[-4mm]
      \begin{align*}
        r_P(t) &= \frac{\sum_i n_i(t{-}1)\,P_i(t)}{\sum_i n_i(t{-}1)\,P_i(t{-}1)}-1\\
        \mathrm{NAV}(t) &= 100\textstyle\prod_{s\le t}\bigl(1+r_P(s)\bigr)
      \end{align*}
      \defl{acheter plus ne change pas $r_P$ : on mesure la \emph{qualité des choix}, pas la taille des apports. Winsorisation $\pm50\,\%$/jour.}\\[1.5mm]
      \begin{tabular}{@{}l r@{}}
        \toprule
        Les 266 portefeuilles & médiane\\\midrule
        CAGR & 12,8\,\%\\
        Volatilité & 21,6\,\%\\
        Sharpe & 0,71\\\bottomrule
      \end{tabular}
    \end{column}
    \begin{column}{0.54\textwidth}\centering
      \includegraphics[width=\linewidth,height=0.64\textheight,keepaspectratio]{figs_bt/B2_spaghetti.png}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Des portefeuilles concentrés : une grosse position domine}
  {\small $\displaystyle w_i(t)=\frac{n_i(t)\,P_i(t)}{V(t)}\;;\qquad \text{concentration}=\overline{\max_i w_i(t)}$}\hspace{6mm}
  \defl{$w_i$ = poids de la ligne $i$ dans le portefeuille ; on suit la plus grosse ligne, moyennée dans le temps.}\\[1.5mm]
  \centering
  \includegraphics[width=0.9\textwidth,height=0.56\textheight,keepaspectratio]{figs_pop/P5_composition.png}\\[1.5mm]
  \kpi{helec}{$\sim$45\,\%}{la plus grosse position, en moyenne}\hspace{8mm}%
  \kpi{dataC}{en haut}{la valeur ; en bas, le poids de chaque ligne}
\end{frame}

\begin{frame}{Ils gardent leurs positions longtemps}
  {\small $\displaystyle \text{durée}_i \;=\; \delta^{\text{vente}}_i - \delta^{\text{achat}}_i \quad(\text{appariement FIFO})$}\hspace{6mm}
  \defl{on apparie chaque vente au plus ancien achat non soldé (FIFO) ; la durée = l'écart entre les deux dates.}\\[1.5mm]
  \centering
  \includegraphics[width=0.9\textwidth,height=0.56\textheight,keepaspectratio]{figs_pop/P5b_duree.png}\\[1.5mm]
  \kpi{helec}{$\sim$1 an}{avant de revendre (médiane 390 j)}\hspace{8mm}%
  \kpi{dataC}{44\,\%}{des positions ne sont jamais revendues}
\end{frame}

\begin{frame}{Deux mètres : l'écart brut (IR) et l'alpha ajusté (appraisal)}
  \begin{columns}[T]
    \begin{column}{0.46\textwidth}
      \small
      \textbf{1 · L'écart brut au marché} (suppose $\beta=1$) :\\[-4mm]
      \begin{align*}
        x_t=r_P(t)-r_{\mathrm{SPY}}(t),\qquad
        \mathrm{IR}=\frac{\bar x}{\sigma(x)}\sqrt{252}
      \end{align*}
      \textbf{2 · Le talent, marché retiré} :\\[-4mm]
      \begin{align*}
        r_P(t)&=\alpha+\beta\,r_{\mathrm{SPY}}(t)+\varepsilon_t\\
        \mathrm{appraisal}&=\frac{\alpha}{\sigma(\varepsilon)}\sqrt{252}
      \end{align*}
      \defl{si $\beta=1$ les deux coïncident ; sinon l'IR mélange \emph{exposition} ($\beta$) et \emph{talent} ($\alpha$).}\\[1.5mm]
      \choixbox{on classe à l'appraisal — ne pas confondre « avoir de l'exposition » et « avoir raison ».}
    \end{column}

\begin{frame}{On choisit qui copier chaque année, sans regarder le futur}
  \begin{columns}[T]
    \begin{column}{0.42\textwidth}
      \small
      \textbf{Sélection point-in-time} à chaque coupe $t^R_Y$ (dernier jour coté de l'année $Y$) :\\[-4mm]
      \begin{align*}
        \mathrm{Top}\text{-}K(Y)=\mathrm{argtop}_K\;
        \mathrm{score}_m\bigl(\text{passé}\le t^R_Y\bigr)
      \end{align*}
      \defl{le score (IR ou appraisal) d'une année n'utilise \emph{que} le passé — walk-forward, aucun regard sur le futur.}\\[3mm]
      \defl{la figure : combien de membres « significatifs » à chaque coupe — de 1 en 2014 à 20 en 2025-26 ; le panier se remplit avec l'historique.}\\[1.5mm]
      \choixbox{aucune donnée future dans la sélection ni dans les poids.}
    \end{column}

\begin{frame}{Le portefeuille des 4 premiers, en parts : le moteur}
  \begin{columns}[T]
    \begin{column}{0.46\textwidth}
      \small
      \textbf{Chaque élu est un fonds} au prix $\mathrm{NAV}^k(t)$ ; à la coupe $t^R_Y$ on achète des parts :\\[-4mm]
      \begin{align*}
        \mathrm{nbr}^k_Y=\frac{w_k\,W(t^R_Y)}{\mathrm{NAV}^k(t^R_Y)},\qquad w_k=\tfrac1K
      \end{align*}
      \textbf{Entre deux coupes} : buy \& hold, les poids dérivent :\\[-4mm]
      \begin{align*}
        r(t)=\sum_k w^k(t{-}1)\,r_k(t)
      \end{align*}
      \defl{$W$ = richesse au rebalancement ($W_0=100$) ; $w^k(t{-}1)$ = poids \emph{effectifs de la veille}.}\\[1.5mm]
      \textbf{Évaluation} par année civile :
      $x_Y=r_Y-r_Y^{\mathrm{SPY}}$,\;
      $t=\tfrac{\bar x}{\sigma(x)}\sqrt{n_{\text{ans}}}\lessgtr 1{,}81$
    \end{column}
    \begin{column}{0.52\textwidth}\centering
      \includegraphics[width=\linewidth,height=0.56\textheight,keepaspectratio]{figs_bt/E2_nav_top4.png}\\[1.5mm]
      \kpi{helec}{+3,1\,\%/an}{excès du top-4 (sél. IR)}\;
      \kpi{warnorange}{$\beta$ 1,20 · $\alpha\approx0$}{porté par le marché}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Combien copier ? le choix de $K$, lui aussi sans tricher}
  \begin{columns}[T]
    \begin{column}{0.40\textwidth}
      \small
      \textbf{$K$ choisi chaque année sur le passé seulement} :\\[-4mm]
      \begin{align*}
        K^\star(Y)=\underset{K\in[4,20]}{\arg\max}\;
        \mathrm{score}\bigl(\text{strat-}K\big|_{\le Y}\bigr)
      \end{align*}
      \defl{même logique walk-forward que la sélection des membres : le $K$ joué en $Y{+}1$ ne voit que le passé.}\\[3mm]
      \kpi{helec}{t 1,58}{meilleur cas du parcours ($K$ walk-forward, sél. IR, $+4{,}1\,\%$/an)}\\[2mm]
      \kpi{grisC}{t 0,14}{la cadence 6 mois n'apporte rien}\\[2mm]
      \defl{$\beta$ de cette config = 1,17 — à ne pas confondre avec le 1,20 du top-4 fixe.}
    \end{column}

  \end{columns}
\end{frame}

\begin{frame}{On change la répartition du capital — rien ne décroche}

\begin{frame}{Autre règle : classer au Sharpe brut, sans test}

      \begin{align*}
        S_m=\frac{\bar r_m}{\sigma(r_m)}\sqrt{252}\;>\;0{,}4
